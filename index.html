The Solution

TrueCost Dynamic Router

An algorithmic gateway. Scores complexity, projects bloat-inclusive cost on each model, routes to the cheapest. One function. Drop-in.

truecost-router.ts
/** TrueCost Dynamic Router · arxiv.org/abs/2603.23971 */ interface ModelConfig { id: string; pricePerMTok: number; bloatExponent: number; // exponential growth (small) bloatLinear: number; // linear growth (large) } // ── Complexity Estimator ───────────── function scoreComplexity(prompt: string): number { let score = Math.min(prompt.length / 500, 3); [/\bprove\b/i, /\bsynthesize\b/i, /\bcalculate\b/i, /\bderive\b/i, /\boptimize\b/i, /\balgorithm\b/i] .forEach(p => { if (p.test(prompt)) score += 1.5; }); score += (prompt.match(/\?/g) || []).length * 0.8; if (/\b(proof|theorem|integral)\b/i.test(prompt)) score += 2.0; return Math.min(score, 10); } // ── Cost Projector ─────────────────── function projectCost( m: ModelConfig, c: number, baseTokens = 2000 ): number { const bloat = m.bloatExponent > 0 ? 1 + c ** m.bloatExponent // small: exponential : 1 + c * m.bloatLinear; // large: linear, stable return (baseTokens * bloat / 1_000_000) * m.pricePerMTok; } // ── Router ─────────────────────────── const MODELS: ModelConfig[] = [ { id: "gemini-3-flash", pricePerMTok: 0.22, bloatExponent: 2.5, bloatLinear: 0 }, { id: "gpt-5.2", pricePerMTok: 1.00, bloatExponent: 0, bloatLinear: 0.2 }, ]; export function routePrompt(prompt: string) { const c = scoreComplexity(prompt); const ranked = MODELS .map(m => ({ m, cost: projectCost(m, c) })) .sort((a, b) => a.cost - b.cost); return { model: ranked[0].m.id, complexityScore: c, projectedCost: ranked[0].cost, reasoning: `c=${c.toFixed(2)} → ${ranked[0].m.id}`, }; }
