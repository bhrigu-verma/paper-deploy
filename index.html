The Solution

TrueCost
Dynamic Router

An algorithmic gateway that intercepts every prompt, estimates complexity, and routes to the statistically cheapest model — after projecting thinking-token bloat. One function. Infinite savings.

truecost-router.ts
/** * TrueCost Dynamic Router * Complexity-aware LLM gateway. * Paper: arxiv.org/abs/2603.23971 */ interface ModelConfig { id: string; pricePerMTok: number; bloatExponent: number; // exponential bloatLinear: number; // linear } interface RouteDecision { model: string; complexityScore:number; projectedCost: number; bloatMultiplier:number; reasoning: string; } // ─── Complexity Estimator ──────────────── function scoreComplexity( prompt: string ): number { let score = 0; // Length signal (0–3) score += Math.min(prompt.length / 500, 3); // Reasoning keywords → +1.5 each [/\bprove\b/i, /\bsynthesize\b/i, /\bcalculate\b/i, /\bderive\b/i, /\boptimize\b/i, /\balgorithm\b/i, /\bstep.by.step\b/i] .forEach(p => { if (p.test(prompt)) score += 1.5; }); // Multi-part queries score += (prompt.match(/\?/g) || []).length * 0.8; // Domain boost: math/science if (/\b(math|proof|theorem|integral)\b/i .test(prompt)) score += 2.0; return Math.min(score, 10); } // ─── Cost Projector ────────────────────── function projectCost( m: ModelConfig, c: number, tokens: number ): number { const bloat = m.bloatExponent > 0 ? 1 + (c ** m.bloatExponent) : 1 + (c * m.bloatLinear); return (tokens * bloat / 1_000_000) * m.pricePerMTok; } // ─── Router ────────────────────────────── const MODELS: ModelConfig[] = [ { id: "gemini-3-flash", pricePerMTok: 0.22, bloatExponent: 2.5, bloatLinear: 0 }, { id: "gpt-5.2", pricePerMTok: 1.00, bloatExponent: 0, bloatLinear: 0.2 }, ]; export function routePrompt( prompt: string ): RouteDecision { const c = scoreComplexity(prompt); const costs = MODELS .map(m => ({ m, cost: projectCost(m, c, 2000) })) .sort((a, b) => a.cost - b.cost); return { model: costs[0].m.id, complexityScore: c, projectedCost: costs[0].cost, bloatMultiplier: costs[0].cost / (costs[0].m.pricePerMTok * 0.002), reasoning: `c=${c.toFixed(2)} → ${costs[0].m.id}`, }; }
